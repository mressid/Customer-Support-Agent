\chapter{Actors and use cases}

\section{Human actors}
\begin{table}[htbp]
\centering\caption{Human actors}\label{tab:actors}
\begin{tabular}{@{}p{3.8cm}p{12.2cm}@{}}
\toprule \textbf{Actor} & \textbf{Role and goals} \\ \midrule
\textbf{Customer} & Asks a question or requests an action through a supported channel (web chat, then voice); wants a correct answer or a human. \\
\textbf{Support agent} & Reviews, takes over and resolves conversations or tickets; receives the handoff context. \\
\textbf{Support lead} & Manages queues, policies, reporting and quality; filters and follows tickets. \\
\textbf{Organization administrator} & Configures channels, knowledge, AI agents, integrations and access; invites users and assigns roles. \\
\textbf{Platform operator} & Monitors reliability, cost, abuse and platform health. \\
\bottomrule
\end{tabular}
\end{table}

\section{System actors}
\begin{table}[htbp]
\centering\caption{External systems and technical actors}\label{tab:systems}
\begin{tabular}{@{}p{4.6cm}p{11.4cm}@{}}
\toprule \textbf{System} & \textbf{Interaction} \\ \midrule
Model provider (LLM) & Generation, classification, embeddings, through an adapter. \\
Speech provider (STT/TTS or speech-to-speech) & Transcription and speech synthesis (R2). \\
Channel providers & Webhooks and outbound delivery (e-mail, messaging; R3). \\
Business systems (commerce, CRM, etc.) & Tool calls through connectors, with limited permissions. \\
Secret manager & Reference and rotation of credentials. \\
Telemetry stack & Logs, metrics, traces, alerts. \\
\bottomrule
\end{tabular}
\end{table}
