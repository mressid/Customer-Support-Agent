% ---------- true landscape pages (header and footer follow the page) ----------
% pdflscape only rotates the viewing angle, which leaves the header on the side.
% Here the paper itself becomes landscape, so header, footer and page number sit
% at the top and bottom of the landscape page. (\newgeometry cannot change paper size.)
\makeatletter
\newdimen\lsmargin \lsmargin=1.8cm
\newcommand{\setphysicalpagesize}{%
  \ifdefined\pdfpagewidth
    \pdfpagewidth=\paperwidth \pdfpageheight=\paperheight
  \else\ifdefined\pagewidth
    \pagewidth=\paperwidth \pageheight=\paperheight
  \fi\fi}
\newcommand{\lsresetvsize}{\global\@colht\textheight \global\@colroom\@colht \global\vsize\@colroom}
\newcommand{\lsbegin}{%
  \clearpage
  \edef\lsrestore{\paperwidth=\the\paperwidth\relax \paperheight=\the\paperheight\relax
    \textwidth=\the\textwidth\relax \textheight=\the\textheight\relax
    \oddsidemargin=\the\oddsidemargin\relax \evensidemargin=\the\evensidemargin\relax
    \topmargin=\the\topmargin\relax}%
  \paperwidth=297mm \paperheight=210mm
  \textwidth=\dimexpr\paperwidth-2\lsmargin\relax
  \textheight=\dimexpr\paperheight-2\lsmargin\relax
  \oddsidemargin=\dimexpr\lsmargin-1in-\hoffset\relax
  \evensidemargin=\oddsidemargin
  \topmargin=\dimexpr\lsmargin-1in-\voffset-\headheight-\headsep\relax
  \onecolumn
  \setlength{\headwidth}{\textwidth}%
  \lsresetvsize
  \setphysicalpagesize}
\newcommand{\lsend}{%
  \clearpage \lsrestore \onecolumn
  \setlength{\headwidth}{\textwidth}%
  \lsresetvsize
  \setphysicalpagesize}
\makeatother
\newenvironment{lscapepage}{\lsbegin}{\lsend}

% ---------- diagrams ----------
% Portrait figure: \mmdfig{path}{caption}{label}
\newcommand{\mmdfig}[3]{%
  \begin{figure}[!htbp]
    \centering
    \includegraphics[max width=\linewidth,max height=0.78\textheight,keepaspectratio]{figures/#1}
    \caption{#2}\label{#3}
  \end{figure}}
% Landscape page for wide diagrams: \mmdfigL{path}{caption}{label}
\newcommand{\mmdfigL}[3]{%
  \begin{lscapepage}
    \centering
    \includegraphics[max width=\linewidth,max height=0.80\textheight,keepaspectratio]{figures/#1}
    \captionof{figure}{#2}\label{#3}
  \end{lscapepage}}

% Landscape page that also carries the preceding headings / story card: \mmdfigLH{pre}{path}{caption}{label}
\newcommand{\mmdfigLH}[4]{%
  \begin{lscapepage}
    #1
    \centering
    \includegraphics[max width=\linewidth,max height=0.68\textheight,keepaspectratio]{figures/#2}
    \captionof{figure}{#3}\label{#4}
  \end{lscapepage}}
